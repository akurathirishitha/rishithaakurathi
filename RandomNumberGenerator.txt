//
int ButtonPressed = 0;

int CanGenerate = 0;

int RandomNumber = 0;

int counter;

int ledArray[6] = {13,12,11,10,9,8};
void setup()
{
    pinMode(0, INPUT);
  	Serial.begin(9600);
  	for(int i = 0; i < 6; i++)
    {
      pinMode(ledArray[i], OUTPUT);
    }
}


void blinkLedS()
{
  for (counter = 0; counter < RandomNumber; ++counter) 
  {
      digitalWrite(ledArray[counter], HIGH);
  }
  delay(1000);
  for (counter = 0; counter < RandomNumber; ++counter) 
  {
      digitalWrite(ledArray[counter], LOW);
  }
  delay(1000);
}

void loop()
{
  ButtonPressed = digitalRead(0);
  if (ButtonPressed == HIGH && CanGenerate == LOW) 
  {
    RandomNumber = random(1, 6 + 1);
    Serial.println(RandomNumber);
    CanGenerate = HIGH;
    blinkLedS();
    blinkLedS();
    blinkLedS();
    CanGenerate = LOW;
  }

}
